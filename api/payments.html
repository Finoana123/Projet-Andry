import { createClient } from '@supabase/supabase-js'; import jwt from 'jsonwebtoken'; const supabase = createClient(process.env.SUPABASE_URL, process.env.SUPABASE_ANON_KEY); function authenticate(req) { const authHeader = req.headers.authorization; if (!authHeader || !authHeader.startsWith('Bearer ')) throw new Error('Non autorisé'); const token = authHeader.split(' ')[1]; return jwt.verify(token, process.env.JWT_SECRET); } export default async function handler(req, res) { try { authenticate(req); } catch (err) { return res.status(401).json({ message: err.message }); } // GET : lister les impayés pour un mois donné (ou mois courant) if (req.method === 'GET' && req.query?.status === 'unpaid') { const month = req.query.month || new Date().toISOString().slice(0,7); // ex: "2026-05" // Tous les élèves const { data: allStudents, error: studentsError } = await supabase .from('students') .select('id, first_name, last_name, matricule, niveau'); if (studentsError) return res.status(500).json({ message: studentsError.message }); // Élèves ayant payé pour ce mois const { data: paid, error: paymentsError } = await supabase .from('payments') .select('student_id') .eq('month', month) .eq('paid', true); if (paymentsError) return res.status(500).json({ message: paymentsError.message }); const paidIds = paid.map(p => p.student_id); const unpaid = allStudents.filter(s => !paidIds.includes(s.id)); return res.status(200).json(unpaid); } // POST : enregistrer un paiement (marquer payé) if (req.method === 'POST') { const { student_id, month, amount } = req.body; if (!student_id || !month) return res.status(400).json({ message: 'student_id et month requis' }); // Vérifier si un enregistrement existe déjà const { data: existing } = await supabase .from('payments') .select('id') .eq('student_id', student_id) .eq('month', month) .maybeSingle(); if (existing) { const { error } = await supabase .from('payments') .update({ paid: true, payment_date: new Date().toISOString(), amount: amount || 0 }) .eq('id', existing.id); if (error) return res.status(500).json({ message: error.message }); } else { const { error } = await supabase .from('payments') .insert([{ student_id, month, paid: true, payment_date: new Date().toISOString(), amount: amount || 0 }]); if (error) return res.status(500).json({ message: error.message }); } return res.status(200).json({ success: true }); } return res.status(405).json({ message: 'Méthode non autorisée' }); }
